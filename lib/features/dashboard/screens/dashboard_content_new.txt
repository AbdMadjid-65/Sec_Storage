class _StoragePieChartPainter extends CustomPainter {
  @override
  void paint(Canvas canvas, Size size) {
    final center = Offset(size.width / 2, size.height / 2);
    
    final bgPaint = Paint()
      ..color = PriVaultColors.surface
      ..style = PaintingStyle.stroke
      ..strokeWidth = 28;
    canvas.drawCircle(center, 70, bgPaint);

    void drawArc(double r, double strokeW, List<Color> colors, double startAngle, double sweepAngle) {
      final rect = Rect.fromCircle(center: center, radius: r);
      final gradient = LinearGradient(colors: colors).createShader(rect);
      final paint = Paint()
        ..shader = gradient
        ..style = PaintingStyle.stroke
        ..strokeWidth = strokeW
        ..strokeCap = StrokeCap.round;
      canvas.drawArc(rect, startAngle, sweepAngle, false, paint);
    }

    final rad = 3.14159 / 180;
    
    drawArc(84, 8, [Colors.pink, Colors.pinkAccent], -90 * rad, 180 * rad);
    drawArc(73, 8, [Colors.amber, Colors.orangeAccent], 45 * rad, 136 * rad);
    drawArc(62, 8, [Colors.lightGreen, Colors.greenAccent], 180 * rad, 90 * rad);
    drawArc(51, 8, [Colors.cyan, Colors.blueAccent], 270 * rad, 108 * rad);
  }

  @override
  bool shouldRepaint(covariant CustomPainter oldDelegate) => false;
}
